在 `src/models.py` 的每个 dataclass 中添加 `to_dict()` 方法，Trace 添加 `to_json()`：

```python
from dataclasses import dataclass
from typing import Any
import json as _json


@dataclass
class Action:
    type: str
    params: dict[str, Any]
    raw: str

    def to_dict(self) -> dict:
        return {"type": self.type, "params": self.params}


@dataclass
class ToolResult:
    action: Action
    success: bool
    output: str
    error: str | None
    duration_ms: int

    def to_dict(self) -> dict:
        return {
            "success": self.success,
            "output": self.output,
            "error": self.error,
            "duration_ms": self.duration_ms,
        }


@dataclass
class GuardrailDecision:
    allowed: bool
    reason: str
    rule_name: str
    requires_approval: bool

    def to_dict(self) -> dict:
        return {
            "allowed": self.allowed,
            "reason": self.reason,
            "rule_name": self.rule_name,
            "requires_approval": self.requires_approval,
        }


@dataclass
class FeedbackSignal:
    status: str
    summary: str
    details: list[dict]
    suggested_action: str | None

    def to_dict(self) -> dict:
        return {
            "status": self.status,
            "summary": self.summary,
            "details": self.details,
            "suggested_action": self.suggested_action,
        }


@dataclass
class Step:
    index: int
    action: Action
    guardrail: GuardrailDecision
    result: ToolResult | None
    feedback: FeedbackSignal | None
    timestamp: float

    def to_dict(self) -> dict:
        return {
            "index": self.index,
            "action": self.action.to_dict(),
            "guardrail": self.guardrail.to_dict(),
            "result": self.result.to_dict() if self.result else None,
            "feedback": self.feedback.to_dict() if self.feedback else None,
            "timestamp": self.timestamp,
        }


@dataclass
class Trace:
    task: str
    steps: list[Step]
    start_time: float
    end_time: float | None

    def to_dict(self) -> dict:
        return {
            "task": self.task,
            "steps": [s.to_dict() for s in self.steps],
            "start_time": self.start_time,
            "end_time": self.end_time,
        }

    def to_json(self, indent: int = 2) -> str:
        return _json.dumps(self.to_dict(), ensure_ascii=False, indent=indent)


@dataclass
class Message:
    role: str
    content: str
```

修改 `src/loop.py`：

```python
import time
import json
import logging
from src.models import Message, Step, Trace
from src.llm.base import LLMClient
from src.tools.registry import ToolRegistry
from src.guardrails.chain import GuardrailChain
from src.actions import ActionParser
from src.feedback import FeedbackParser
from src.memory import Memory

logger = logging.getLogger(__name__)


class AgentLoop:
    def __init__(
        self,
        llm: LLMClient,
        tools: ToolRegistry,
        guardrails: GuardrailChain,
        parser: ActionParser,
        feedback: FeedbackParser,
        memory: Memory,
        project_dir: str,
        max_steps: int = 20,
        hitl=None,
    ):
        self._llm = llm
        self._tools = tools
        self._guardrails = guardrails
        self._parser = parser
        self._feedback = feedback
        self._memory = memory
        self._project_dir = project_dir
        self._max_steps = max_steps
        self._hitl = hitl

    async def run(self, task: str) -> Trace:
        trace = Trace(task=task, steps=[], start_time=time.time(), end_time=None)
        messages: list[Message] = []

        # 构建初始上下文
        messages.append(Message(role="system", content=self._build_system_prompt()))
        messages.append(Message(role="user", content=task))

        logger.info("AgentLoop 开始: task=%s, max_steps=%d", task, self._max_steps)

        for step_index in range(self._max_steps):
            # 1. 调用 LLM（带重试）
            llm_text = await self._call_llm_with_retry(messages)

            # 2. 解析动作
            action = self._parser.parse(llm_text, self._tools.tool_names())

            timestamp = time.time()

            # 3. 护栏检查
            guardrail = self._guardrails.check(action, self._project_dir)

            if not guardrail.allowed:
                logger.warning("护栏拦截: step=%d, action=%s, rule=%s, reason=%s",
                               step_index, action.type, guardrail.rule_name, guardrail.reason)
                step = Step(
                    index=step_index,
                    action=action,
                    guardrail=guardrail,
                    result=None,
                    feedback=None,
                    timestamp=timestamp,
                )
                trace.steps.append(step)
                self._memory.add_step(step)
                messages.append(Message(role="assistant", content=llm_text))
                messages.append(Message(
                    role="system",
                    content=f"[护栏拦截] {guardrail.reason}。请尝试其他方式。",
                ))
                continue

            if guardrail.requires_approval:
                approved = False
                if self._hitl is not None:
                    approved = self._hitl.prompt(action, guardrail)

                if not approved:
                    logger.info("HITL 拒绝: step=%d, action=%s, rule=%s",
                                step_index, action.type, guardrail.rule_name)
                    step = Step(
                        index=step_index,
                        action=action,
                        guardrail=guardrail,
                        result=None,
                        feedback=None,
                        timestamp=timestamp,
                    )
                    trace.steps.append(step)
                    self._memory.add_step(step)
                    messages.append(Message(role="assistant", content=llm_text))
                    reject_reason = "用户拒绝" if self._hitl else "非交互模式下自动拒绝"
                    messages.append(Message(
                        role="system",
                        content=f"[护栏需审批] {guardrail.reason}。动作未执行（{reject_reason}）。",
                    ))
                    continue
                logger.info("HITL 批准: step=%d, action=%s", step_index, action.type)

            # 4. 执行工具
            if action.type == "stop":
                logger.info("AgentLoop 停机: step=%d, summary=%s",
                            step_index, action.params.get("summary", ""))
                step = Step(
                    index=step_index,
                    action=action,
                    guardrail=guardrail,
                    result=None,
                    feedback=None,
                    timestamp=timestamp,
                )
                trace.steps.append(step)
                self._memory.add_step(step)
                break

            result = self._tools.dispatch(action, self._project_dir)

            # 5. 解析反馈
            fb = self._feedback.parse(result)

            logger.debug("步骤完成: step=%d, action=%s, success=%s, feedback=%s",
                         step_index, action.type, result.success, fb.status)

            # Memory 追踪
            if fb.status in ("fail", "error"):
                self._memory.increment_failure()
            if action.type == "write_file" and result.success:
                self._memory.mark_file_modified(action.params.get("path", ""))

            # 6. 记录 step
            step = Step(
                index=step_index,
                action=action,
                guardrail=guardrail,
                result=result,
                feedback=fb,
                timestamp=timestamp,
            )
            trace.steps.append(step)
            self._memory.add_step(step)

            # 7. 构建下一轮反馈
            output_safe = (result.output or "")[:500]
            messages.append(Message(role="assistant", content=llm_text))
            messages.append(Message(
                role="system",
                content=f"""
[步骤结果]
动作: {action.type}({json.dumps(action.params)})
成功: {result.success}
输出: {output_safe}
错误: {result.error or '无'}
反馈: status={fb.status}, summary={fb.summary}, suggested_action={fb.suggested_action or '无'}
""",
            ))

        trace.end_time = time.time()
        logger.info("AgentLoop 结束: steps=%d, duration=%.2fs",
                     len(trace.steps), trace.end_time - trace.start_time)
        return trace

    async def _call_llm_with_retry(self, messages: list[Message]) -> str:
        """调用 LLM，最多重试 1 次（空响应或网络错误）。"""
        last_error = None
        for attempt in range(2):
            try:
                llm_text = await self._llm.chat(messages)
            except Exception as e:
                last_error = str(e)
                logger.warning("LLM 调用失败 (attempt=%d/%d): %s",
                               attempt + 1, 2, last_error)
                if attempt == 0:
                    continue
                raise RuntimeError(f"LLM 调用失败，已重试: {last_error}") from e

            if llm_text and llm_text.strip():
                return llm_text

            logger.warning("LLM 返回空响应 (attempt=%d/%d)", attempt + 1, 2)
            if attempt == 0:
                messages.append(Message(
                    role="system",
                    content="你的上一次响应为空。请提供一个有效的 JSON 动作。",
                ))

        return '{"type": "stop", "params": {"summary": "LLM 连续返回空响应，强制停机"}}'

    def _build_system_prompt(self) -> str:
        tool_list = "\n".join(f"- {name}" for name in sorted(self._tools.tool_names()))
        return f"""你是一个 coding agent。你可以使用以下工具：

{tool_list}

请用 JSON 格式回复动作：
{{"type": "<工具名>", "params": {{...}}}}

要停止时使用：
{{"type": "stop", "params": {{"summary": "你做了什么"}}}}
"""
```

```python
from src.models import Action, Trace
from src.llm.mock import ScriptedMockLLM
from src.tools.registry import ToolRegistry
from src.tools.files import read_file, write_file
from src.tools.shell import shell_handler
from src.guardrails.chain import GuardrailChain
from src.guardrails.rules import (
    PathSandbox, DangerousCmd, NetworkGate, FileWriteGate, DestructiveOp
)
from src.guardrails.hitl import HITLConsole
from src.actions import ActionParser
from src.feedback import FeedbackParser
from src.memory import Memory
from src.loop import AgentLoop
from src.config import Config


class REPL:
    def __init__(self, project_dir: str = "."):
        self._project_dir = project_dir
        self._config = Config.load("config.yaml")
        self._running = False
        self._last_trace: Trace | None = None

    def run(self) -> None:
        self._running = True
        print("Coding Agent Harness REPL")
        print("输入任务描述开始，或 !help 查看命令。")
        print()

        while self._running:
            try:
                line = input("harness> ").strip()
            except (EOFError, KeyboardInterrupt):
                print()
                break

            if not line:
                continue

            if line.startswith("!"):
                self._handle_command(line)
            else:
                self._handle_task(line)

    def _handle_command(self, line: str) -> None:
        parts = line.split()
        cmd = parts[0].lower()

        if cmd == "!help":
            print("命令列表:")
            print("  !help        - 显示帮助")
            print("  !trace       - 查看最近一次 trace 摘要")
            print("  !step N      - 查看第 N 步详情")
            print("  !config      - 查看当前配置")
            print("  !exit        - 退出 REPL")
            print()
            print("或直接输入任务描述来启动 agent。")
        elif cmd == "!exit":
            self._running = False
        elif cmd == "!config":
            import json
            print(json.dumps(self._config, indent=2, default=str))
        elif cmd == "!trace":
            if self._last_trace is None:
                print("暂无 trace（请先运行一个任务）")
            else:
                self._print_trace_summary(self._last_trace)
        elif cmd == "!step":
            if len(parts) < 2:
                print("用法: !step <步骤编号>")
                return
            try:
                step_index = int(parts[1])
            except ValueError:
                print(f"无效的步骤编号: {parts[1]}")
                return
            if self._last_trace is None:
                print("暂无 trace（请先运行一个任务）")
            else:
                self._print_step_detail(self._last_trace, step_index)
        else:
            print(f"未知命令: {cmd}")

    def _handle_task(self, task: str) -> None:
        print(f"\n[Agent] 开始任务: {task}")
        tools = self._build_tools()
        guardrails = self._build_guardrails()
        llm = self._build_llm()

        parser = ActionParser()
        feedback = FeedbackParser()
        memory = Memory(
            max_recent_steps=self._config["memory"]["max_recent_steps"],
            project_dir=self._config["memory"]["project_dir"],
        )

        loop = AgentLoop(
            llm=llm,
            tools=tools,
            guardrails=guardrails,
            parser=parser,
            feedback=feedback,
            memory=memory,
            project_dir=self._project_dir,
            max_steps=self._config["loop"]["max_steps"],
            hitl=HITLConsole(),
        )

        import asyncio
        trace = asyncio.run(loop.run(task))
        self._last_trace = trace

        # 打印 trace 摘要
        self._print_trace_summary(trace)

    def _print_trace_summary(self, trace: Trace) -> None:
        print(f"\n[Agent] 完成，共 {len(trace.steps)} 步:")
        for step in trace.steps:
            status = "[OK]" if step.guardrail.allowed else "[BLOCKED]"
            if step.guardrail.requires_approval:
                status = "[APPROVAL]"
            print(f"  [{step.index}] {step.action.type} {status}")
            if step.feedback:
                print(f"       -> {step.feedback.status}: {step.feedback.summary}")
        print()

    def _print_step_detail(self, trace: Trace, step_index: int) -> None:
        if step_index < 0 or step_index >= len(trace.steps):
            print(f"步骤 {step_index} 不存在（共 {len(trace.steps)} 步）")
            return
        step = trace.steps[step_index]
        print(f"\n=== 步骤 [{step.index}] 详情 ===")
        print(f"动作类型: {step.action.type}")
        print(f"动作参数: {step.action.params}")
        print(f"护栏判定: {'放行' if step.guardrail.allowed else '拦截'}")
        if step.guardrail.rule_name:
            print(f"护栏规则: {step.guardrail.rule_name}")
        if step.guardrail.reason:
            print(f"护栏原因: {step.guardrail.reason}")
        if step.guardrail.requires_approval:
            print(f"需审批: 是")
        if step.result:
            print(f"执行结果: {'成功' if step.result.success else '失败'}")
            print(f"输出: {step.result.output[:200]}")
            if step.result.error:
                print(f"错误: {step.result.error}")
            print(f"耗时: {step.result.duration_ms}ms")
        if step.feedback:
            print(f"反馈状态: {step.feedback.status}")
            print(f"反馈摘要: {step.feedback.summary}")
            if step.feedback.suggested_action:
                print(f"建议动作: {step.feedback.suggested_action}")
        print()

    def _build_tools(self) -> ToolRegistry:
        tools = ToolRegistry()
        tools.register("read_file", read_file, {"path": "str"})
        tools.register("write_file", write_file, {"path": "str", "content": "str"})
        tools.register("shell", shell_handler, {"command": "str"})
        tools.register("stop", lambda a, p: None, {"summary": "str?"})
        return tools

    def _build_guardrails(self) -> GuardrailChain:
        chain = GuardrailChain()
        rules_config = self._config["guardrails"]["rules"]
        chain.add_rule(PathSandbox(), enabled=rules_config["path_sandbox"]["enabled"])
        chain.add_rule(DangerousCmd(), enabled=rules_config["dangerous_cmd"]["enabled"])
        chain.add_rule(NetworkGate(), enabled=rules_config["network_gate"]["enabled"])
        chain.add_rule(FileWriteGate(), enabled=rules_config["file_write_gate"]["enabled"])
        chain.add_rule(DestructiveOp(), enabled=rules_config["destructive_op"]["enabled"])
        return chain

    def _build_llm(self):
        provider = self._config["llm"]["provider"]
        if provider == "openai":
            try:
                from src.llm.openai import OpenAIClient
                return OpenAIClient(
                    model=self._config["llm"]["model"],
                    base_url=self._config["llm"]["base_url"],
                    max_tokens=self._config["llm"]["max_tokens"],
                )
            except ImportError:
                pass
        return ScriptedMockLLM([Action(type="stop", params={"summary": "未配置 LLM"}, raw="")])
```

```python
import time
from pathlib import Path
from src.models import Step


class Memory:
    def __init__(self, max_recent_steps: int = 5, project_dir: str = ".harness/memory"):
        self._steps: list[Step] = []
        self._max_recent_steps = max_recent_steps
        self._project_dir = Path(project_dir)
        self._modified_files: set[str] = set()
        self._failure_count = 0

    # ── 会话内记忆 ──

    def add_step(self, step: Step) -> None:
        self._steps.append(step)
        if len(self._steps) > self._max_recent_steps:
            self._steps = self._steps[-self._max_recent_steps:]

    def get_recent_steps(self, n: int | None = None) -> list[Step]:
        if n is None:
            n = self._max_recent_steps
        return self._steps[-n:]

    @property
    def failure_count(self) -> int:
        return self._failure_count

    def increment_failure(self) -> None:
        self._failure_count += 1

    def reset_failure_count(self) -> None:
        self._failure_count = 0

    def mark_file_modified(self, filepath: str) -> None:
        self._modified_files.add(filepath)

    def is_file_modified(self, filepath: str) -> bool:
        return filepath in self._modified_files

    # ── 项目级记忆：约定 ──

    def write_convention(self, key: str, content: str) -> None:
        self._project_dir.mkdir(parents=True, exist_ok=True)
        filepath = self._project_dir / f"{key}.md"
        filepath.write_text(content, encoding="utf-8")

    def get_convention(self, key: str) -> str | None:
        filepath = self._project_dir / f"{key}.md"
        if filepath.exists():
            return filepath.read_text(encoding="utf-8")
        return None

    def load_conventions(self) -> str:
        """加载默认约定文件 conventions.md。"""
        filepath = self._project_dir / "conventions.md"
        if filepath.exists():
            return filepath.read_text(encoding="utf-8")
        return ""

    # ── 项目级记忆：决策 ──

    def record_decision(self, summary: str, detail: str) -> None:
        """记录一条架构/设计决策到 decisions/ 目录。"""
        decisions_dir = self._project_dir / "decisions"
        decisions_dir.mkdir(parents=True, exist_ok=True)
        timestamp = int(time.time())
        filename = f"{timestamp}-{self._slugify(summary[:30])}.md"
        content = f"""# 决策: {summary}

**时间:** {time.strftime('%Y-%m-%d %H:%M:%S', time.localtime(timestamp))}

{detail}
"""
        (decisions_dir / filename).write_text(content, encoding="utf-8")

    # ── 项目级记忆：失败记录 ──

    def record_failure(self, step_desc: str, fix_desc: str) -> None:
        """记录一次失败及其修复方案到 failures/ 目录。"""
        failures_dir = self._project_dir / "failures"
        failures_dir.mkdir(parents=True, exist_ok=True)
        timestamp = int(time.time())
        filename = f"{timestamp}-{self._slugify(step_desc[:30])}.md"
        content = f"""# 失败记录

**时间:** {time.strftime('%Y-%m-%d %H:%M:%S', time.localtime(timestamp))}

## 失败描述
{step_desc}

## 修复方案
{fix_desc}
"""
        (failures_dir / filename).write_text(content, encoding="utf-8")

    # ── 检索 ──

    def search(self, query: str) -> list[str]:
        results = []
        if self._project_dir.exists():
            for filepath in self._project_dir.rglob("*.md"):
                try:
                    content = filepath.read_text(encoding="utf-8")
                    if query.lower() in content.lower():
                        results.append(content)
                except Exception:
                    pass
        return results

    @staticmethod
    def _slugify(text: str) -> str:
        """将文本转为文件名安全的 slug。"""
        import re
        text = re.sub(r'[^\w\s-]', '', text)
        text = re.sub(r'[-\s]+', '-', text)
        return text.strip('-').lower() or "record"
```
